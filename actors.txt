CREATE TABLE IF NOT EXISTS "actors"
(
id INTEGER PRIMARY KEY AUTOINCREMENT,
age INTEGER,
sex TEXT CHECK(sex = "male" or sex = "female"),
country TEXT,
name TEXT NOT NULL,
last_name TEXT NOT NULL
)

UPDATE actors
SET country = "Unknown"
WHERE country IS NULL

UPDATE actors
SET age = age + 2
WHERE sex = "female"

SELECT name, last_name, sex, country, age
FROM actors
WHERE age < 35 AND country = "France" AND sex = "female"

SELECT name, last_name, sex, country, age
FROM actors
ORDER BY age ASC, name ASC
LIMIT 5

SELECT 
count(*) AS "Общее количество актёров",
avg(age) AS "Средний возраст",
min(age) AS "Минимальный возраст",
max(age) as "Максимальный возраст"
FROM actors

SELECT country
FROM actors
GROUP BY country
HAVING count(*) > 5 AND count(*) < 10

SELECT sex, avg(age) AS "Средний возраст"
FROM actors
GROUP BY sex

SELECT name, last_name, sex, country, min(age) 
FROM actors
GROUP BY country

SELECT DISTINCT country
FROM actors
WHERE country NOT IN(
	SELECT country
	FROM actors
	WHERE sex = "female"
	)
ORDER BY country
